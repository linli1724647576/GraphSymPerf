\begin{algorithm}[H]
\caption{Pattern embedding into an AST}
\label{alg:pattern-embedding}
\begin{algorithmic}[1]
\Require Abstract AST $A_0$, typed hole $h$, Pattern $p$
\Ensure Embedded AST $A$, mutation site $s$, record $M$, or rejection
\If{$\neg$\Call{Compatible}{$h,p$}}\Comment{Interface categories and types}
\State \Return \textsc{Reject}\Comment{Incompatible Pattern}
\EndIf
\State $(K,\tau,\Phi,\sigma,\epsilon)\gets$\Call{Expand}{$p,h$}\Comment{Local source, rewrite, and premises}
\State $(K,\tau,\Phi)\gets$\Call{AlphaRename}{$K,\tau,\Phi,\text{\textsc{Vars}}(A_0)$}\Comment{Avoid capture; retain imports}
\State $A\gets$\Call{Replace}{$A_0,h,K$}\Comment{Preserve the outer context}
\State $s\gets$\Call{Relocate}{$h,\sigma$}\Comment{Actual mutation site}
\State $e\gets$\Call{Relocate}{$h,\epsilon$}\Comment{Complete semantic boundary}
\State $M\gets$\Call{Register}{$p,s,e,\tau,\Phi$}\Comment{Premises passed to M3}
\If{$\neg$\Call{ValidateEmbedding}{$A_0,A,h,M$}}\Comment{Types, scope, and unchanged host}
\State \Return \textsc{Reject}\Comment{Invalid embedding}
\EndIf
\State \Call{FreezeStructure}{$A,M$}\Comment{Only declared structural rewrites}
\State \Return $(A,s,M)$\Comment{Symbols remain unbound}
\end{algorithmic}
\end{algorithm}
