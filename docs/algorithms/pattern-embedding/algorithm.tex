\begin{algorithm}[H]
\caption{Pattern embedding into an AST}
\label{alg:pattern-embedding}
\begin{algorithmic}[1]
\Require Abstract AST \textit{ast} with one typed hole; predefined Pattern \textit{pattern}
\Ensure AST \textit{tree} with the Pattern embedded, or rejection
\State \textit{slot} $\gets$ \Call{FindHole}{\textit{ast}}\Comment{The hole reserved by M1}
\If{not \Call{Compatible}{\textit{slot}, \textit{pattern}}}\Comment{Check interface and output type}
\State \Return \textsc{Reject}
\EndIf
\State \textit{fragment} $\gets$ \Call{ExpandPattern}{\textit{pattern}, \textit{slot}}\Comment{Choose the local structure}
\State Rename private variables in \textit{fragment} to avoid name conflicts
\State \textit{tree} $\gets$ \Call{FillHole}{\textit{ast}, \textit{slot}, \textit{fragment}}\Comment{Preserve the outer context}
\State Mark the mutation site inside the inserted \textit{fragment}
\State Save the mutation rule and its full enclosing context\Comment{Used later in M5 and M6}
\State Save the pattern conditions for graph construction\Comment{Checked later in M3}
\If{not \Call{CheckEmbedding}{\textit{ast}, \textit{tree}, \textit{slot}}}\Comment{Types, scope, and insertion range}
\State \Return \textsc{Reject}
\EndIf
\State Freeze the structure except for the saved mutation\Comment{Keep terminal placeholders}
\State \Return \textit{tree}
\end{algorithmic}
\end{algorithm}
