### Describe the situation

Hi, JanusGraph developers,

Please consider these two queries:

```groovy
// q1
(g.inject(1).V().hasLabel('ScorRoot').has('id',P.eq(_gsp_p0)).as('_gsp_v0').outE('GSP_SC_HIGH').inV().hasLabel('ScorTarget').as('_gsp_v1').filter(__.constant(_gsp_p1).is(P.lt(_gsp_p2))).project('c0').by(__.constant(_gsp_p3))).limit(10001).toList()
```

```groovy
// q2
(g.inject(1).V().hasLabel('ScorRoot').has('id',P.eq(_gsp_p0)).as('_gsp_v0').outE('GSP_SC_REF').inV().hasLabel('ScorTarget').as('_gsp_v1').filter(__.constant(_gsp_p1).is(P.lt(_gsp_p2))).project('c0').by(__.constant(_gsp_p3))).limit(10001).toList()
```

Send both queries through the Gremlin HTTP endpoint with these integer bindings:

```json
{"_gsp_p0":1,"_gsp_p1":2,"_gsp_p2":1,"_gsp_p3":1}
```

The two traversals differ only in the outgoing edge label. Both return no rows because the child comparison is `2 < 1`, regardless of the target vertex. On the graph below, the median paired q1/q2 runtime ratio is about 70.21.

def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False, allow_nan=False)


def sha256(value):
    return hashlib.sha256(canonical(value).encode()).hexdigest()


def require(condition, message):
    if not condition:
        raise ValueError(message)


def typed(value):
    if value is None:
        return ("null",)
    if type(value) in (bool, int, str):
        return (type(value).__name__, value)
    if type(value) is float and math.isfinite(value):
        return ("float", value)
    if type(value) is list:
        return ("list", tuple(typed(item) for item in value))
    if type(value) is dict and all(type(key) is str for key in value):
        return ("map", tuple(sorted((key, typed(item)) for key, item in value.items())))
    raise ValueError("Unsupported diagnostic/result value type: " + type(value).__name__)


def bag(rows):
    return Counter(typed(row) for row in rows)


def decode(value):
    """The adapter's finite scalar/list/map GraphSON contract, without coercion."""
    if type(value) in (str, bool, int) or value is None:
        return value
    if type(value) is float and math.isfinite(value):
        return value
    if type(value) is list:
        return [decode(item) for item in value]
    require(type(value) is dict, "Unsupported GraphSON value")
    if set(value) == {"@type", "@value"}:
        kind, raw = value["@type"], value["@value"]
        if kind in ("g:Int32", "g:Int64", "gx:BigInteger") and type(raw) is int:
            return raw
        if kind in ("g:Float", "g:Double") and type(raw) in (int, float) and math.isfinite(raw):
            return float(raw)
        if kind == "g:List" and type(raw) is list:
            return [decode(item) for item in raw]
        if kind == "g:Map" and type(raw) is list and len(raw) % 2 == 0:
            result = {}
            for key, item in zip(raw[::2], raw[1::2]):
                key = decode(key)
                require(type(key) is str and key not in result, "Invalid GraphSON map key")
                result[key] = decode(item)
            return result
        raise ValueError("Unsupported GraphSON type: " + str(kind))
    require(all(type(key) is str for key in value), "Invalid JSON map key")
    return {key: decode(item) for key, item in value.items()}


def decode_data(envelope, limit=None):
    data = envelope.get("result", {}).get("data")
    if type(data) is dict and data.get("@type") == "g:List":
        data = data.get("@value")
    require(type(data) is list, "HTTP result data must be a list")
    rows = []
    for item in data:
        if type(item) is dict and item.get("@type") == "g:Traverser":
            raw = item.get("@value", {})
            bulk = decode(raw.get("bulk"))
            require(type(bulk) is int and 0 < bulk <= 10_000_001, "Invalid traverser bulk")
            count = min(bulk, limit - len(rows)) if limit is not None else bulk
            rows.extend([decode(raw.get("value"))] * count)
        else:
            rows.append(decode(item))
        if limit is not None and len(rows) >= limit:
            break
    return rows


def at(value, index):
    if type(value) is dict:
        require(set(value) == {"_gsp_affine"}, "Invalid affine tag")
        base, step = value["_gsp_affine"]
        require(type(base) is type(step) is int, "Invalid affine coefficients")
        return base + step * index
    return value


def iter_nodes():
    for group in PAYLOAD["node_groups"]:
        for index in range(group["count"]):
            properties = {key: at(value, index) for key, value in group["properties"].items()}
            yield {"uid": properties["uid"], "labels": group["labels"], "properties": properties}


def iter_edges():
    ordinal = 0
    for group in PAYLOAD["edge_groups"]:
        for index in range(group["count"]):
            yield {"source": at(group["source"], index), "target": at(group["target"], index),
                   "source_label": group["source_label"], "target_label": group["target_label"],
                   "type": group["type"], "ordinal": ordinal,
                   "properties": {key: at(value, index) for key, value in group["properties"].items()}}
            ordinal += 1


def batches(rows):
    iterator = iter(rows)
    while batch := list(islice(iterator, 250)):
        yield batch


class HTTP:
    def __init__(self, args):
        parsed = urlsplit(args.url)
        require(parsed.scheme in ("http", "https") and parsed.hostname and not
                (parsed.username or parsed.password or parsed.query or parsed.fragment), "Invalid HTTP endpoint")
        require(all(re.fullmatch(r"[A-Za-z0-9_\-]+", name) for name in
                    (args.graph_binding, args.traversal_source)), "Invalid graph aliases")
        self.connection = (HTTPSConnection if parsed.scheme == "https" else HTTPConnection)(
            parsed.hostname, parsed.port, timeout=30.0)
        self.endpoint = parsed.path.rstrip("/") or "/"
        self.aliases = {"g": args.traversal_source, "_gsp_graph": args.graph_binding}
        self.headers = {"Content-Type": "application/json", "Accept": PAYLOAD["mime"]}
        if args.user:
            require(args.password_env and args.password_env in os.environ, "Password environment variable is required")
            self.headers["Authorization"] = "Basic " + base64.b64encode(
                (args.user + ":" + os.environ[args.password_env]).encode()).decode()
        self.requests = 0

    def call(self, script, bindings=None, *, raw=False, limit=None):
        payload = {"gremlin": script, "language": "gremlin-groovy", "bindings": bindings or {}, "aliases": self.aliases}
        if self.connection.sock is None:
            self.connection.connect()
        else:
            self.connection.sock.settimeout(30.0)
        started = perf_counter()
        self.requests += 1
        self.connection.request("POST", self.endpoint, body=json.dumps(payload, allow_nan=False).encode(), headers=self.headers)
        response = self.connection.getresponse()
        body = response.read()
        if response.getheader("Content-Encoding", "").lower() == "gzip":
            body = gzip.decompress(body)
        envelope = json.loads(body)
        code = envelope.get("status", {}).get("code")
        if response.status >= 300 or code not in (200, 204):
            raise RuntimeError(canonical({"http_status": response.status, "status": envelope.get("status"),
                                          "request_number": self.requests, "request_retried": False}))
        if code == 204:
            envelope["result"] = {"data": [], "meta": {}}
        elapsed = (perf_counter() - started) * 1000
        return (envelope if raw else decode_data(envelope, limit)), elapsed

    def close(self):
        self.connection.close()


def check_bag(actual, expected, phase):
    require(bag(actual) == bag(expected), canonical({"phase": phase, "reason": "typed_bags_differ",
            "actual_count": len(actual), "expected_count": len(expected), "actual_first": actual[:3], "expected_first": expected[:3]}))


def setup(http, emit):
    for number, operation in enumerate(PAYLOAD["setup_operations"]):
        rows, _ = http.call(operation["script"], operation["bindings"])
        check_bag(rows, operation["expected_rows"], operation["kind"])
        emit({"phase": "setup_schema", "operation": operation["kind"], "number": number, "status": "passed"})
    for kind, iterator, script in (("nodes", iter_nodes(), PAYLOAD["vertex_script"]),
                                   ("edges", iter_edges(), PAYLOAD["edge_script"])):
        processed = 0
        for number, batch in enumerate(batches(iterator)):
            values, _ = http.call(script, {"gsp_rows": batch})
            check_bag(values, [len(batch)], "load_" + kind)
            processed += len(batch)
            emit({"phase": "setup_load", "kind": kind, "batch": number, "batch_size": len(batch), "processed": processed})
        require(processed == PAYLOAD["counts"][kind], "Loaded count differs from GraphSpec")
    for operation in PAYLOAD["index_checks"]:
        rows, _ = http.call(operation["script"])
        check_bag(rows, operation["expected_rows"], "native_index_check")
    emit({"phase": "setup", "status": "passed", "counts": PAYLOAD["counts"], "batch_size": 250})


def logical_properties(properties, *, vertex=False):
    inverse = PAYLOAD["physical_to_logical_properties"]
    ignored = {"_gsp_identity"} if vertex else set()
    result = {}
    for physical, value in properties.items():
        if physical in ignored:
            continue
        logical = inverse.get(physical, physical)
        require(logical not in result, "Native property names collided")
        result[logical] = value
    return result


def verify_graph(http, emit):
    """Read every identity in bounded batches, each edge through its target once."""
    scripts = PAYLOAD["verification_scripts"]
    native_nodes, native_edges = set(), set()

        for row in actual:
            require(type(row.get("native_id")) is str and row["native_id"] not in native_nodes, "Duplicate native vertex ID")

    # Reject any unexpected duplicate target before issuing a diagnostic query.
